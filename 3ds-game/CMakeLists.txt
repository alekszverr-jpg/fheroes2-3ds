cmake_minimum_required(VERSION 3.24)
project(fheroes2_3ds LANGUAGES C CXX)
if(NOT NINTENDO_3DS)
    message(FATAL_ERROR "Use cmake/native-3ds.cmake")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(USE_SDL_VERSION SDL2)
set(ENABLE_IMAGE OFF)
set(MACOS_APP_BUNDLE OFF)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TEST OFF CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_SENSOR OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/vendor/SDL2" sdl2)
foreach(feature INSTALL SAMPLES DEPS_SHARED CMD MOD MIDI OPUS WAVPACK)
    set(SDL2MIXER_${feature} OFF CACHE BOOL "" FORCE)
endforeach()
# Embedded decoders for a first build. Hardware audio/performance still needs testing.
set(SDL2MIXER_VORBIS STB CACHE STRING "" FORCE)
add_subdirectory("${ROOT}/vendor/SDL2_mixer" mixer)
add_library(SDL2_mixer::SDL2_mixer ALIAS SDL2_mixer)
add_library(threads_3ds INTERFACE)
target_link_libraries(threads_3ds INTERFACE ctru)
add_library(Threads::Threads ALIAS threads_3ds)
find_package(ZLIB REQUIRED)
set(GNU_CC_WARN_OPTS -Wall -Wextra)
set(GNU_CXX_WARN_OPTS -Wall -Wextra)
add_compile_options(-fsigned-char)
add_subdirectory("${ROOT}/fheroes2/src/thirdparty" thirdparty)
add_subdirectory("${ROOT}/fheroes2/src/engine" engine)
file(GLOB_RECURSE GAME_SOURCES CONFIGURE_DEPENDS "${ROOT}/fheroes2/src/fheroes2/*.cpp")
add_executable(fheroes2-3ds ${GAME_SOURCES})
foreach(directory agg ai army audio battle campaign castle dialog editor game gui h2d heroes image kingdom maps monster resource spell system world)
    target_include_directories(fheroes2-3ds PRIVATE "${ROOT}/fheroes2/src/fheroes2/${directory}")
endforeach()
target_compile_definitions(fheroes2-3ds PRIVATE FHEROES2_DATA=sdmc:/3ds/fheroes2)
target_link_libraries(fheroes2-3ds PRIVATE SDL2::SDL2main engine)
find_program(THREEDSXTOOL 3dsxtool HINTS "${PROBE_SDK}/tools/bin" REQUIRED)
find_program(SMDHTOOL smdhtool HINTS "${PROBE_SDK}/tools/bin" REQUIRED)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/fheroes2.smdh"
    COMMAND "${SMDHTOOL}" --create "fheroes2 3DS preview" "Main menu and native two-screen renderer" "fheroes2 contributors / 3DS port"
        "${PROBE_SDK}/libctru/default_icon.png" "${CMAKE_CURRENT_BINARY_DIR}/fheroes2.smdh"
    DEPENDS "${PROBE_SDK}/libctru/default_icon.png" VERBATIM)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/fheroes2.3dsx"
    COMMAND "${THREEDSXTOOL}" "$<TARGET_FILE:fheroes2-3ds>" "${CMAKE_CURRENT_BINARY_DIR}/fheroes2.3dsx"
        "--smdh=${CMAKE_CURRENT_BINARY_DIR}/fheroes2.smdh"
    DEPENDS fheroes2-3ds "${CMAKE_CURRENT_BINARY_DIR}/fheroes2.smdh" VERBATIM)
add_custom_target(game_3dsx ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/fheroes2.3dsx")
add_executable(fheroes2-engine-test engine_smoke.cpp)
foreach(directory agg ai army audio battle campaign castle dialog editor game gui h2d heroes image kingdom maps monster resource spell system world)
    target_include_directories(fheroes2-engine-test PRIVATE "${ROOT}/fheroes2/src/fheroes2/${directory}")
endforeach()
target_link_libraries(fheroes2-engine-test PRIVATE SDL2::SDL2main engine)
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/fheroes2-engine-test.3dsx"
    COMMAND "${THREEDSXTOOL}" "$<TARGET_FILE:fheroes2-engine-test>" "${CMAKE_CURRENT_BINARY_DIR}/fheroes2-engine-test.3dsx"
    DEPENDS fheroes2-engine-test VERBATIM)
add_custom_target(engine_test_3dsx ALL DEPENDS "${CMAKE_CURRENT_BINARY_DIR}/fheroes2-engine-test.3dsx")
